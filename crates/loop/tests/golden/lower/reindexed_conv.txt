kernel conv(%x: i32[128] read, %w: i32[144] read, %out: i32[128] write)

  %n_p_q.0 = blockIdx.y
  %n_p_q.1 = threadIdx.y
  %m.0 = blockIdx.x
  %m.1 = threadIdx.x
  alloc %acc: i32[1] local {
    %acc[0] = 0:i32
    alloc %x_tile: i32[16] shared {
      alloc %w_tile: i32[16] shared {
        for %r_s_c.0 in 9 {
          barrier
          if %n_p_q.1 * 4 + %m.1 < 4 {
            %x_tile[(%n_p_q.1 * 4 + %m.1) * 4] <- async %x[floormod(%n_p_q.0, 4) * 16 + %m.1 * 4 + floormod(%r_s_c.0, 3) * 4 + floordiv(%n_p_q.0 * 4 + %m.1, 16) * 64 + floordiv(%r_s_c.0 * 4, 12) * 16 + -20] run 4 if %m.1 + floormod(%r_s_c.0, 3) + -1 < 4, %m.1 * -1 + floormod(%r_s_c.0, 3) * -1 + 1 < 1, floormod(%n_p_q.0, 4) + floordiv(%r_s_c.0 * 4, 12) + -1 < 4, floormod(%n_p_q.0, 4) * -1 + floordiv(%r_s_c.0 * 4, 12) * -1 + 1 < 1

          }
          async commit
          async wait 0
          barrier
          for %r_s_c.1 in 4 {
            %acc[0] = add(%acc[0], mul(%x_tile[%n_p_q.1 * 4 + %r_s_c.1], %w_tile[%m.1 * 4 + %r_s_c.1]))
          }
        }
      }
    }
    %out[%n_p_q.1 * 4 + floordiv(%n_p_q.1 + %n_p_q.0 * 4, 16) * 64 + floormod(%n_p_q.0, 4) * 16 + %m.1 + %m.0 * 4] = %acc[0]
  }
