kernel row_sums(%x: i32[8] read, %out: i32[4] write)
  grid(1, 1, 1) block(3, 1, 1) smem 0
  %i.1 = threadIdx.x
  alloc %acc: i32[1] local {
    %acc[0] = 0:i32
    for %k in 2 {
      %acc[0] = add(%acc[0], %x[%k + %i.1 * 2])
    }
    %out[%i.1] = %acc[0]
  }
  alloc %acc: i32[1] local {
    %acc[0] = 0:i32
    for %k in 2 {
      if %i.1 + 3 < 4 {
        %acc[0] = add(%acc[0], %x[%k + %i.1 * 2 + 6])
      }
    }
    if %i.1 + 3 < 4 {
      %out[%i.1 + 3] = %acc[0]
    }
  }
